import {chromium} from 'playwright';
import {readFile,writeFile} from 'node:fs/promises';
import assert from 'node:assert/strict';
const out='/home/cjjnbb/projects/rtl-architect/docs/evidence/storage-budget',open=JSON.parse(await readFile('/tmp/rtlarch-budget-browser/open.json','utf8'));
const browser=await chromium.launch({headless:true,args:['--no-sandbox']}),page=await browser.newPage({viewport:{width:1600,height:1100}}),errors=[];page.on('pageerror',e=>errors.push(String(e)));
try{
 const url=new URL(open.reviewUrl);url.searchParams.set('panel','work');await page.goto(url.toString());await page.getByRole('button',{name:'方案预算',exact:true}).click();await page.locator('.budget-history > summary').click();await page.locator('.react-flow__node').first().waitFor();
 const target=page.locator('.budget-history').getByText(/BANK_MAPPING_UNKNOWN/);await target.evaluate(el=>el.scrollIntoView({block:'center'}));await page.screenshot({path:out+'/saved-budget.png'});
 await page.setViewportSize({width:1100,height:820});await target.evaluate(el=>el.scrollIntoView({block:'center'}));await page.screenshot({path:out+'/narrow-budget.png'});assert(await page.evaluate(()=>document.documentElement.scrollWidth<=innerWidth));assert.deepEqual(errors,[]);
 await writeFile(out+'/browser-snapshot-followup.json',JSON.stringify({checkedAt:new Date().toISOString(),buildId:open.service.buildId,readOnly:true,graphReady:true,errors},null,2)+'\n');
}finally{await browser.close();}
