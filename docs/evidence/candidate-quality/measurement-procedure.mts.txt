import {readFile,writeFile} from 'node:fs/promises';
import {modelSchema,layoutSchema} from '/home/cjjnbb/projects/rtl-architect/src/core/schema.ts';
import {modelHash} from '/home/cjjnbb/projects/rtl-architect/src/core/export.ts';
import {prepareDiagram} from '/home/cjjnbb/projects/rtl-architect/src/web/graph/prepareDiagram.ts';
import {diagramQuality} from '/home/cjjnbb/projects/rtl-architect/src/web/graph/quality.ts';
import {diagramSvg} from '/home/cjjnbb/projects/rtl-architect/src/web/graph/exportSvg.ts';
const phase=process.argv[2],reports=[];
for(const example of ['typed-expressions','external-axis']){
 const model=modelSchema.parse(JSON.parse(await readFile(`examples/${example}/architecture/model.json`,'utf8'))),hash=modelHash(model);
 let layout;try{layout=layoutSchema.parse(JSON.parse(await readFile(`examples/${example}/architecture/layout.json`,'utf8')));}catch{}
 for(const scope of model.modules.filter(m=>m.kind==='subsystem').map(m=>m.id))for(const task of [undefined,{kind:'overview' as const},...model.localSignals!.filter(s=>s.scope===scope&&s.driver.kind==='register').slice(0,1).map(s=>({kind:'state' as const,stateId:s.id}))]){
 const start=performance.now(),{graph,positions}=await prepareDiagram(model,scope,hash,layout?.scopes[scope],{task}),elapsedMs=performance.now()-start;
 const id=`${example}-${scope}-${task?.kind??'full'}`;
 const report={id,hash,task:task??null,elapsedMs,nodes:graph.nodes.length,edges:graph.edges.length,quality:diagramQuality(graph)};reports.push(report);
 await writeFile(`docs/evidence/candidate-quality/${phase}-${id}.json`,JSON.stringify({report,graph,positions},null,2));
 await writeFile(`docs/evidence/candidate-quality/${phase}-${id}.svg`,diagramSvg(model,scope,{task,positions,showPortNames:true},hash,graph));
 console.log(JSON.stringify({id,ms:Math.round(elapsedMs),crossings:report.quality.crossings,hidden:report.quality.hiddenLabels.length}));
 }
}
await writeFile(`docs/evidence/candidate-quality/${phase}.json`,JSON.stringify(reports,null,2));
